% FIGURE 1 (fig:CPS): a Clifford-enhanced Product State (CPS) and its use
% for computation. Introduction.
% Standalone: compile from this folder (pdflatex CEPS-MSI-2.tex).
% Included in main.tex at 0.8\linewidth.
% ----------------------------------------------------------------------
% Original request (author): redraw the submitted figures in TikZ,
% keeping their content and colours (this one was exported from Inkscape
% in arXiv v2).
% ----------------------------------------------------------------------
% What is drawn, as in the arXiv v2 figure (137 x 71 bp page):
%   - inputs |0>, |0> and the magic state T|+> (light blue tag, label
%     "Magic State") entering a Clifford circuit;
%   - a layer of single-qubit Pauli measurements ("Pauli meas.").
% Coordinates are in bp, origin at the top-left corner, y pointing down,
% read from the arXiv v2 PDF, so the figure keeps its size and layout.
% Same drawing as figure MSI of arXiv v1 (MSI.tex in the history), without
% the dashed CPS box and the green measurement panel.
% ======================================================================
\documentclass[tikz,border=0pt]{standalone}
\input{figpreamble}
\figsansfonts

% Colours of the arXiv v2 figure.
\definecolor{cpsbluetext}{RGB}{67,91,236}   % "Magic State"
\definecolor{cpsbluefill}{RGB}{155,187,255} % tag of the magic state

\tikzset{
  cps wire/.style={line width=0.5375bp},
  % A measurement: box 18.17 x 14.13 bp, top-left corner at the origin.
  pics/cpsmeter/.style={code={
    \draw[line width=0.977bp, line cap=round, line join=round]
      (0,0) rectangle (18.169,14.132)
      (1.229,10.316) .. controls (5.529,6.015) and (12.494,5.984)
        .. (16.833,10.215)
      (9.083,11.105) -- (14.13,2.019);
  }},
}

\begin{document}
\begin{tikzpicture}[x=1bp, y=-1bp, every node/.style={inner sep=0pt}]
  \useasboundingbox (0,0) rectangle (137,71);

  % ---------------- circuit ----------------
  % The strokes of the arXiv v2 PDF carry an x-scale of 0.988318.
  \begin{scope}[xscale=0.988318]
    \foreach \y in {9.578, 30.406, 51.336}
      \draw[cps wire] (37.133,\y) -- (59.938,\y);
    \foreach \y in {9.309, 30.324, 51.336}
      \draw[cps wire] (90.344,\y) -- (113.15,\y);
    \draw[line width=0.415bp] (59.907,0.207) rectangle ++(30.469,62.566);
    \pic at (113.399,1.973) {cpsmeter};
    \pic at (113.328,22.977) {cpsmeter};
    \pic at (113.478,44.539) {cpsmeter};
  \end{scope}

  % ---------------- magic-state tag (blue, over its wire) ----------------
  \fill[cpsbluefill, opacity=0.377839, rounded corners=2.379bp]
    (0,41.809) rectangle (56.348,60.812);

  % ---------------- labels ----------------
  \node[font=\figsize{14.59}] at (25.56,9.36) {$\ket{0}$};
  \node[font=\figsize{14.21}] at (25.47,30.42) {$\ket{0}$};
  \node[font=\figsize{14.43}] at (16.74,51.84) {$T\ket{+}$};
  \node[font=\figsize{7.68}] at (74.79,30.33) {Clifford};
  \node[font=\figsize{9.16}] at (29.70,66.78) {\color{cpsbluetext}\bfseries Magic State};
  \node[font=\figsize{7.63}] at (118.53,65.79) {Pauli meas.};
\end{tikzpicture}
\end{document}
