% What is drawn, as in the submitted figure (251 x 249 bp page):
%   - top: rho, the circuit C^dagger, then a Pauli measurement P in P_n

%   - bottom: rho measured directly in CPC^dagger (pink panel), which is
%     in P_n if C is Clifford;
%   - right: Tr[C^dagger rho C P] = Tr[rho CPC^dagger], with P and
%     CPC^dagger tinted as their panels; thin arrows link each panel to
%     its side of the equation.
% Coordinates are in bp, origin at the top-left corner, y pointing down,
% read from the submitted PDF, so the figure keeps its size and layout.
% Serif fonts, as in the submitted figure.

\input{figpreamble}

% Colours of the submitted figure.

  bp grey/.style={fill=bpgrey, fill opacity=0.509558},
  bp pink/.style={fill=bppink, fill opacity=0.328077},

  % A measurement: box 17.97 x 13.98 bp, top-left corner at the origin.
  pics/bpmeter/.style={code={
    \draw[line width=0.97bp, line cap=round, line join=round]
      (0,0) rectangle (17.97,13.98)
      (1.215,10.207) .. controls (5.467,5.953) and (12.357,5.918)
        .. (16.648,10.105)
      (8.984,10.984) -- (13.975,2.0);
  }},
  % Thin black arrows from a panel to its side of the equation.
  bp link/.style={line width=#1, -{Triangle[length=3.6bp, width=4bp]}},
}
