% read from the submitted PDF, so the figure keeps its size and layout.

\documentclass[tikz,border=0pt]{standalone}
\input{figpreamble}

\tikzset{
  msi wire/.style={line width=0.53bp},
  % A measurement: box 18 x 14 bp, top-left corner at the origin.
  pics/msimeter/.style={code={
    \draw[line width=0.97bp, line cap=round, line join=round]
      (0,0) rectangle (18,14)

      (9.0,11.0) -- (14.0,2.0);
  }},
}
